// =====================================================================
// 3D ADVANCED INDUSTRIAL PLANT LAYOUT & PROCESS SIMULATOR
// Features: Random Layout -> Tweak Layout -> Real/Optimized Layout 
//           -> Process Steps Table -> Flow Process Summary Chart
// =====================================================================

int currentMode = 1; // 1: Random 3D, 2: Tweak 3D, 3: Real Optimized 3D, 4: Steps Chart, 5: Summary Report
int activeTab = 3;   // Active table tab (1: Random, 2: Tweak, 3: Real Optimized)

float totalDistanceWalked = 0;
int totalTimeSpent = 0;
boolean isComplete = false;

ArrayList<Machine> machines = new ArrayList<Machine>();
ArrayList<Room> rooms = new ArrayList<Room>();
Worker worker;

// Camera System
PVector camEye, camTarget;
PVector currentCamEye, currentCamTarget;
float worldSize = 1400;

// Interactive Tweak Variable
int selectedMachineIndex = -1;

void setup() {
  fullScreen(P3D);
  orientation(LANDSCAPE);
  
  camEye = new PVector(0, -1200, 1000);
  camTarget = new PVector(0, 0, 0);
  currentCamEye = camEye.copy();
  currentCamTarget = camTarget.copy();
  
  buildLayout();
}

void draw() {
  background(15, 20, 25);
  
  if (currentMode <= 3) {
    // --- 3D ENVIRONMENT RENDERING ---
    currentCamEye.lerp(camEye, 0.05);
    currentCamTarget.lerp(camTarget, 0.05);
    
    camera(currentCamEye.x, currentCamEye.y, currentCamEye.z,
           currentCamTarget.x, currentCamTarget.y, currentCamTarget.z,
           0, 1, 0);
           
    ambientLight(140, 140, 150);
    directionalLight(255, 255, 255, 0.6, 1, -0.4);
    pointLight(200, 220, 255, 0, -800, 0);
    
    drawFloor();
    
    for (Room r : rooms) r.draw();
    for (int i = 0; i < machines.size(); i++) {
      machines.get(i).draw(i == selectedMachineIndex);
    }
    
    // Draw Travel Route Path
    stroke(255, 150, 50, 180);
    strokeWeight(4);
    for (int i = 0; i < machines.size() - 1; i++) {
      line(machines.get(i).opX, 5, machines.get(i).opZ, 
           machines.get(i+1).opX, 5, machines.get(i+1).opZ);
    }
    
    worker.update();
    worker.draw();
  }
  
  // --- 2D OVERLAYS / HUD / CHARTS ---
  hint(DISABLE_DEPTH_TEST);
  camera(); 
  noLights();
  
  if (currentMode <= 3) {
    drawHUD();
  } else if (currentMode == 4) {
    drawSequenceStepsChart();
  } else if (currentMode == 5) {
    drawSummaryReportChart();
  }
  
  hint(ENABLE_DEPTH_TEST);
}

// =========================================================
// INTERACTION & LAYOUT BUILDER
// =========================================================

void mousePressed() {
  float btnW = 320, btnH = 65;
  float btnX = width - btnW - 20, btnY = 20;
  
  // Navigation Button
  if (mouseX > btnX && mouseX < btnX + btnW && mouseY > btnY && mouseY < btnY + btnH) {
    currentMode++;
    if (currentMode > 5) currentMode = 1;
    buildLayout();
    return;
  }
  
  // Mode 2: Interactive Machine Tweaking (Click to swap machine layout positions)
  if (currentMode == 2) {
    for (int i = 0; i < machines.size(); i++) {
      Machine m = machines.get(i);
      // Rough screen distance check for selecting machine node
      if (dist(mouseX, mouseY, width/2 + m.x*0.3, height/2 + m.z*0.3) < 60) {
        if (selectedMachineIndex == -1) {
          selectedMachineIndex = i;
        } else {
          // Swap positions of two machines
          float tempX = machines.get(selectedMachineIndex).x;
          float tempZ = machines.get(selectedMachineIndex).z;
          machines.get(selectedMachineIndex).setPosition(m.x, m.z);
          m.setPosition(tempX, tempZ);
          selectedMachineIndex = -1;
        }
        break;
      }
    }
  }
  
  // Modes 4 & 5 Tab Switcher
  if (currentMode >= 4) {
    float tabW = 180, tabH = 45;
    float startTabX = width*0.05;
    float tabY = 80;
    
    for (int i = 1; i <= 3; i++) {
      float tx = startTabX + (i-1)*(tabW + 10);
      if (mouseX > tx && mouseX < tx + tabW && mouseY > tabY && mouseY < tabY + tabH) {
        activeTab = i;
      }
    }
  }
}

void buildLayout() {
  machines.clear();
  rooms.clear();
  totalDistanceWalked = 0;
  totalTimeSpent = 0;
  isComplete = false;
  selectedMachineIndex = -1;
  
  float q = worldSize / 4;
  
  if (currentMode == 1) {
    // ----------------------------------------------------
    // SCREEN 1: RANDOM / PROCESS LAYOUT (Unoptimized Rooms)
    // ----------------------------------------------------
    rooms.add(new Room(-q, -q, q*2, q*2, "Raw Material Store", color(40, 50, 60)));
    rooms.add(new Room(q, -q, q*2, q*2, "Cutting Dept", color(50, 40, 60)));
    rooms.add(new Room(-q, q, q*2, q*2, "Machining Dept", color(60, 50, 40)));
    rooms.add(new Room(q, q, q*2, q*2, "Quality Inspection", color(40, 60, 50)));
    
    machines.add(new Machine(-q*1.2, -q*1.2, PI, "Storage"));
    machines.add(new Machine(q*1.2, -q*0.8, PI/2, "Hacksaw"));
    machines.add(new Machine(-q*1.2, q*0.7, 0, "Lathe"));
    machines.add(new Machine(-q*0.6, q*1.3, PI/2, "Milling"));
    machines.add(new Machine(q*1.1, q*1.1, PI, "Inspection"));
    machines.add(new Machine(-q*0.8, -q*0.8, 0, "Storage"));
    
  } else if (currentMode == 2) {
    // ----------------------------------------------------
    // SCREEN 2: TWEAK / ADJUSTABLE LAYOUT (Semi-Optimized)
    // ----------------------------------------------------
    rooms.add(new Room(0, 0, worldSize*1.1, worldSize*0.9, "Flexible Manufacturing Floor", color(45, 50, 55)));
    
    machines.add(new Machine(-q*1.2, -q*0.5, PI/2, "Storage"));
    machines.add(new Machine(-q*0.4, -q*0.8, PI/2, "Hacksaw"));
    machines.add(new Machine(q*0.3, -q*0.2, 0, "Lathe"));
    machines.add(new Machine(-q*0.2, q*0.6, PI, "Milling"));
    machines.add(new Machine(q*0.6, q*0.8, PI/2, "Inspection"));
    machines.add(new Machine(q*1.2, q*0.2, 0, "Storage"));
    
  } else if (currentMode == 3) {
    // ----------------------------------------------------
    // SCREEN 3: REAL / OPTIMIZED PRODUCT LAYOUT (21m Path)
    // ----------------------------------------------------
    float spacing = worldSize / 5.5;
    float startX = -worldSize/2 + spacing/2;
    
    rooms.add(new Room(0, 0, worldSize*1.1, worldSize*0.4, "Optimized Assembly Line Floor", color(35, 55, 65)));
    
    machines.add(new Machine(startX, 0, PI/2, "Storage"));
    machines.add(new Machine(startX + spacing*1, 0, PI/2, "Hacksaw"));
    machines.add(new Machine(startX + spacing*2, 0, PI/2, "Lathe"));
    machines.add(new Machine(startX + spacing*3, 0, PI/2, "Milling"));
    machines.add(new Machine(startX + spacing*4, 0, PI/2, "Inspection"));
    machines.add(new Machine(startX + spacing*5, 0, PI/2, "Storage"));
  }
  
  if (currentMode <= 3) {
    worker = new Worker(machines.get(0).opX, machines.get(0).opZ);
    triggerOutsideView();
  }
  
  if (currentMode >= 4) {
    activeTab = (currentMode == 4) ? 3 : activeTab;
  }
}

// =========================================================
// CAMERAS & ENVIRONMENT
// =========================================================

void triggerInsideView(float mx, float mz, float mRot) {
  float offsetX = sin(mRot) * 250;
  float offsetZ = cos(mRot) * 250;
  camEye.set(mx + offsetX, -180, mz + offsetZ);
  camTarget.set(mx, -60, mz);
}

void triggerOutsideView() {
  camEye.set(0, -1100, 900);
  camTarget.set(0, 0, 0);
}

void drawFloor() {
  pushMatrix();
  rotateX(HALF_PI);
  fill(30);
  noStroke();
  rectMode(CENTER);
  rect(0, 0, worldSize*3, worldSize*3);
  
  stroke(45);
  strokeWeight(2);
  for(float i = -worldSize * 1.5; i <= worldSize * 1.5; i += 100) {
    line(i, -worldSize*1.5, i, worldSize*1.5);
    line(-worldSize*1.5, i, worldSize*1.5, i);
  }
  popMatrix();
}

// =========================================================
// HUD & CHARTS

void drawHUD() {
  fill(15, 20, 30, 240);
  noStroke();
  rectMode(CORNER);
  rect(0, 0, width, 120);
  
  fill(255);
  textAlign(LEFT, TOP);
  textSize(28);
  
  if (currentMode == 1) text("SCREEN 1: RANDOM / PROCESS LAYOUT (Unoptimized)", 20, 15);
  else if (currentMode == 2) text("SCREEN 2: TWEAK LAYOUT (Interactive Component Drag/Swap)", 20, 15);
  else if (currentMode == 3) text("SCREEN 3: REAL / OPTIMIZED PRODUCT LAYOUT (Ideal Line)", 20, 15);
  
  fill(0, 255, 150);
  textSize(24);
  text("Distance Walked: " + nf(totalDistanceWalked, 0, 1) + "m  |  Time Elapsed: " + totalTimeSpent + " min", 20, 65);
  
  fill(255, 200, 0);
  textSize(24);
  if (worker != null) text(worker.statusMsg, 20, height - 50);
  
  String nextBtnLabel = "";
  if (currentMode == 1) nextBtnLabel = "Go to Tweak Mode";
  else if (currentMode == 2) nextBtnLabel = "Go to Real Optimized";
  else if (currentMode == 3) nextBtnLabel = "View Steps Table";
  
  drawSwitchButton(nextBtnLabel);
}

// SCREEN 4: SEQUENCE OF STEPS TABLE (LAB RECORD)
void drawSequenceStepsChart() {
  background(240, 245, 250); 
  
  fill(30);
  textAlign(CENTER, TOP);
  textSize(32);
  text("SEQUENCE OF STEPS (FLOW PROCESS CHART)", width/2, 20);
  
  drawTabs();
  
  float tableW = width * 0.90;
  float tableH = height * 0.68;
  float tX = width * 0.05;
  float tY = 135;
  
  stroke(0);
  strokeWeight(2);
  noFill();
  rectMode(CORNER);
  rect(tX, tY, tableW, tableH);
  
  // Header
  fill(220, 230, 240);
  rect(tX, tY, tableW, 45);
  
  line(tX, tY + 45, tX + tableW, tY + 45);
  fill(0);
  textSize(20);
  textAlign(CENTER, CENTER);
  text("Step", tX + 50, tY + 22);
  text("Type", tX + 180, tY + 22);
  text("Description", tX + width*0.42, tY + 22);
  text("Distance", tX + tableW - 220, tY + 22);
  text("Time", tX + tableW - 80, tY + 22);
  
  // Select data based on Active Tab
  String[][] rowData = getSequenceDataForTab(activeTab);
  
  float rowH = (tableH - 90) / 10;
  
  for (int i = 0; i < 10; i++) {
    float y = tY + 45 + (i * rowH);
    line(tX, y, tX + tableW, y);
    
    drawChartIcon(rowData[i][1], tX + 180, y + rowH/2);
    
    fill(0);
    textAlign(CENTER, CENTER);
    text(rowData[i][0], tX + 50, y + rowH/2);
    
    textAlign(LEFT, CENTER);
    text(rowData[i][2], tX + 280, y + rowH/2);
    
    textAlign(CENTER, CENTER);
    fill(200, 50, 50);
    text(rowData[i][3], tX + tableW - 220, y + rowH/2);
    fill(50, 100, 200);
    text(rowData[i][4], tX + tableW - 80, y + rowH/2);
  }
  
  // Total Row
  float totalY = tY + tableH - 45;
  fill(230, 235, 240);
  rect(tX, totalY, tableW, 45);
  fill(0);
  textAlign(CENTER, CENTER);
  text("TOTAL", tX + width*0.42, totalY + 22);
  fill(200, 50, 50); text(getTotalDistForTab(activeTab), tX + tableW - 220, totalY + 22);
  fill(50, 100, 200); text(getTotalTimeForTab(activeTab), tX + tableW - 80, totalY + 22);
  
  drawSwitchButton("View Summary Report");
}

// SCREEN 5: SUMMARY REPORT & FLOW DIAGRAM
void drawSummaryReportChart() {
  background(245, 245, 240);
  
  fill(30);
  textAlign(CENTER, TOP);
  textSize(32);
  text("FLOW PROCESS CHART REPORT & SUMMARY", width/2, 18);
  
  drawTabs();
  
  // --- LEFT SIDE: VERTICAL PROCESS FLOW DIAGRAM ---
  float diaX = width * 0.22;
  float startY = 145;
  float stepGap = 38;
  
  stroke(80);
  strokeWeight(2);
  line(diaX, startY + 10, diaX, startY + 9 * stepGap);
  
  String[] flowTypes = {"Storage", "Transport", "Operation", "Transport", "Operation", "Transport", "Delay", "Operation", "Inspection", "Storage"};
  String[][] flowLabels = getFlowLabelsForTab(activeTab);
  
  for (int i = 0; i < 10; i++) {
    float y = startY + (i * stepGap);
    
    fill(255);
    stroke(0);
    ellipse(diaX, y, 24, 24);
    drawChartIcon(flowTypes[i], diaX + 35, y);
    
    fill(0);
    textAlign(LEFT, CENTER);
    textSize(16);
    text(flowLabels[i][0] + " (" + flowLabels[i][1] + ")", diaX + 70, y);
  }
  
  // --- RIGHT SIDE: LAB RECORD REPORT MATRIX ---
  float tabX = width * 0.54;
  float tabY = 150;
  float tabW = width * 0.41;
  float tabH = 340;
  
  fill(255);
  stroke(0);
  strokeWeight(2);
  rect(tabX, tabY, tabW, tabH);
  
  fill(220, 230, 210);
  rect(tabX, tabY, tabW, 45);
  fill(0);
  textAlign(CENTER, CENTER);
  textSize(18);
  text("Symbol", tabX + 45, tabY + 22);
  text("Activity", tabX + 135, tabY + 22);
  text("Frequency", tabX + 235, tabY + 22);
  text("Time / Dist", tabX + tabW - 60, tabY + 22);
  line(tabX, tabY + 45, tabX + tabW, tabY + 45);
  
  String[][] summaryData = getSummaryMatrixForTab(activeTab);
  
  float sRowH = (tabH - 45) / 5;
  for (int i = 0; i < 5; i++) {
    float y = tabY + 45 + (i * sRowH);
    line(tabX, y, tabX + tabW, y);
    
    drawChartIcon(summaryData[i][0], tabX + 45, y + sRowH/2);
    
    fill(0);
    textAlign(CENTER, CENTER);
    text(summaryData[i][0], tabX + 135, y + sRowH/2);
    text(summaryData[i][1], tabX + 235, y + sRowH/2);
    fill(0, 100, 200);
    text(summaryData[i][2], tabX + tabW - 60, y + sRowH/2);
  }
  
  drawSwitchButton("Restart Simulation");
}

void drawTabs() {
  float tabW = 180, tabH = 40;
  float startTabX = width * 0.05;
  float tabY = 75;
  
  String[] tabTitles = {"1: Random", "2: Tweaked", "3: Real (Optimized)"};
  
  for (int i = 1; i <= 3; i++) {
    float tx = startTabX + (i-1)*(tabW + 10);
    if (activeTab == i) fill(60, 120, 200);
    else fill(200, 210, 220);
    
    stroke(0);
    rectMode(CORNER);
    rect(tx, tabY, tabW, tabH, 6, 6, 0, 0);
    
    if (activeTab == i) fill(255);
    else fill(50);
    textAlign(CENTER, CENTER);
    textSize(18);
    text(tabTitles[i-1], tx + tabW/2, tabY + tabH/2);
  }
}

void drawChartIcon(String type, float x, float y) {
  stroke(0);
  strokeWeight(2);
  noFill();
  pushMatrix();
  translate(x - 20, y);
  
  if (type.equals("Operation")) ellipse(0, 0, 18, 18);
  else if (type.equals("Transport")) { triangle(-8, -8, 8, 0, -8, 8); }
  else if (type.equals("Inspection")) { rectMode(CENTER); rect(0, 0, 18, 18); }
  else if (type.equals("Delay")) { arc(0, 0, 18, 18, -PI/2, PI/2); line(0, -9, -9, -9); line(-9, -9, -9, 9); line(-9, 9, 0, 9); }
  else if (type.equals("Storage")) { triangle(-8, -8, 8, -8, 0, 8); }
  
  popMatrix();
}

void drawSwitchButton(String label) {
  float btnW = 320, btnH = 65;
  float btnX = width - btnW - 20, btnY = 20;
  
  fill(60, 120, 200);
  stroke(255);
  strokeWeight(2);
  rectMode(CORNER);
  rect(btnX, btnY, btnW, btnH, 10);
  
  fill(255);
  textAlign(CENTER, CENTER);
  textSize(22);
  text(label, btnX + btnW/2, btnY + btnH/2);
}

class Room {
  float x, z, w, d;
  String name;
  color col;
  
  Room(float _x, float _z, float _w, float _d, String _name, color _c) {
    x = _x; z = _z; w = _w; d = _d; name = _name; col = _c;
  }
  
  void draw() {
    pushMatrix();
    translate(x, 0, z);
    
    // Floor Tile
    pushMatrix();
    rotateX(HALF_PI);
    fill(col);
    noStroke();
    rectMode(CENTER);
    rect(0, 0, w, d);
    popMatrix();
    
    // Wireframe Boundary
    stroke(150, 180, 220, 80);
    strokeWeight(4);
    noFill();
    translate(0, -50, 0);
    box(w, 100, d);
    
    popMatrix();
  }
}

class Machine {
  float x, z, rot;
  float opX, opZ;
  String type;
  
  Machine(float _x, float _z, float _rot, String _type) {
    type = _type;
    setPosition(_x, _z);
    rot = _rot;
  }
  
  void setPosition(float _x, float _z) {
    x = _x; z = _z;
    opX = x + sin(rot) * 110;
    opZ = z + cos(rot) * 110;
  }
  
  void draw(boolean isSelected) {
    pushMatrix();
    translate(x, -5, z);
    rotateY(rot);
    
    if (isSelected) stroke(0, 255, 255);
    else stroke(30);
    strokeWeight(isSelected ? 3 : 1);
    
    if (type.equals("Storage")) {
      fill(80, 90, 100);
      translate(0, -75, 0);
      pushMatrix(); translate(-75, 0, -35); box(10, 150, 10); popMatrix();
      pushMatrix(); translate(75, 0, -35); box(10, 150, 10); popMatrix();
      pushMatrix(); translate(-75, 0, 35); box(10, 150, 10); popMatrix();
      pushMatrix(); translate(75, 0, 35); box(10, 150, 10); popMatrix();
      fill(120, 130, 140);
      pushMatrix(); translate(0, 60, 0); box(160, 5, 80); popMatrix();
      pushMatrix(); translate(0, 0, 0); box(160, 5, 80); popMatrix();
      pushMatrix(); translate(0, -60, 0); box(160, 5, 80); popMatrix();
      fill(220, 190, 50);
      pushMatrix(); translate(0, -5, 0); box(140, 15, 20); popMatrix();
    } 
    else if (type.equals("Hacksaw")) {
      fill(50, 120, 200);
      translate(0, -35, 0);
      box(100, 60, 70);
      fill(100);
      translate(0, -35, 0);
      box(80, 10, 60);
      fill(200, 50, 50);
      translate(-20, -15, 0);
      rotateZ(PI/8);
      box(120, 15, 15);
      translate(0, -25, 0);
      fill(255);
      box(80, 2, 2);
    } 
    else if (type.equals("Lathe")) {
      fill(50, 130, 90);
      translate(0, -40, 0);
      box(80, 80, 70);
      translate(0, -45, 0);
      fill(180);
      box(180, 15, 60);
      fill(100);
      translate(-70, -35, 0);
      box(60, 60, 70);
      translate(45, 0, 0);
      rotateY(PI/2);
      fill(80);
      drawCylinder(16, 25, 30);
      translate(0, 0, -90);
      fill(120);
      box(35, 45, 35);
      translate(0, 0, 45);
      translate(-25, 0, 0);
      fill(50);
      box(25, 25, 25);
    } 
    else if (type.equals("Milling")) {
      fill(220, 140, 30);
      translate(0, -25, 0);
      box(90, 40, 100);
      translate(0, -80, -30);
      box(60, 140, 60);
      translate(0, -60, 40);
      box(45, 45, 80);
      translate(0, 40, -10);
      fill(180);
      drawCylinder(12, 8, 40);
      translate(0, 45, 0);
      fill(150);
      box(120, 20, 40);
    } 
    else if (type.equals("Inspection")) {
      fill(200);
      translate(0, -50, 0);
      box(140, 8, 90);
      fill(100);
      translate(-60, 25, -35); box(8, 50, 8);
      translate(120, 0, 0); box(8, 50, 8);
      translate(0, 0, 70); box(8, 50, 8);
      translate(-120, 0, 0); box(8, 50, 8);
      
      fill(50, 150, 255);
      translate(60, -30, -35);
      box(40, 2, 25);
      fill(200, 50, 50);
      translate(20, -2, 0);
      box(10, 5, 10);
    }
    
    popMatrix();
    
    // Operating Base Pad
    pushMatrix();
    translate(opX, 2, opZ);
    rotateY(rot);
    fill(255, 255, 0, 80);
    noStroke();
    rectMode(CENTER);
    rotateX(HALF_PI);
    rect(0, 0, 50, 50);
    popMatrix();
  }
}

void drawCylinder(int sides, float r, float h) {
  float angle = TWO_PI / sides;
  beginShape(TRIANGLE_FAN);
  vertex(0, 0, h/2);
  for (int i = 0; i <= sides; i++) vertex(cos(i * angle) * r, sin(i * angle) * r, h/2);
  endShape();
  beginShape(TRIANGLE_FAN);
  vertex(0, 0, -h/2);
  for (int i = 0; i <= sides; i++) vertex(cos(i * angle) * r, sin(i * angle) * r, -h/2);
  endShape();
  beginShape(QUAD_STRIP);
  for (int i = 0; i <= sides; i++) {
    float x = cos(i * angle) * r, y = sin(i * angle) * r;
    vertex(x, y, h/2);
    vertex(x, y, -h/2);
  }
  endShape();
}

// =========================================================
// WORKER AGENT
// =========================================================

class Worker {
  float x, z;
  int targetIndex = 1;
  float speed = 7.0;
  int workTimer = 0;
  String statusMsg = "Initializing...";
  float animPhase = 0;
  
  Worker(float startX, float startZ) {
    x = startX;
    z = startZ;
  }
  
  void update() {
    if (isComplete) return;
    
    Machine target = machines.get(targetIndex);
    
    if (workTimer > 0) {
      workTimer--;
      animPhase += 0.25;
      statusMsg = "WORKING: Operating " + target.type;
      
      if (workTimer == 0) {
        triggerOutsideView();
        
        if (target.type.equals("Hacksaw")) totalTimeSpent += 4;
        else if (target.type.equals("Lathe")) totalTimeSpent += 5;
        else if (target.type.equals("Milling")) totalTimeSpent += 12; // 10 op + 2 delay
        else if (target.type.equals("Inspection")) totalTimeSpent += 2;
        
        if (targetIndex < machines.size() - 1) {
          targetIndex++;
        } else {
          isComplete = true;
          statusMsg = "Task Complete! Tap top right button to proceed.";
        }
      }
      return;
    }
    
    float d = dist(x, z, target.opX, target.opZ);
    statusMsg = "MOVING: Walking to " + target.type + " Station";
    
    if (d > speed) {
      animPhase += 0.35;
      float angle = atan2(target.opZ - z, target.opX - x);
      x += cos(angle) * speed;
      z += sin(angle) * speed;
      
      if (currentMode == 3) {
        // Optimized line path scaled to exact 21m target distance
        totalDistanceWalked += (speed * 0.015);
      } else if (currentMode == 2) {
        totalDistanceWalked += (speed * 0.025);
      } else {
        totalDistanceWalked += (speed * 0.038);
      }
    } else {
      x = target.opX;
      z = target.opZ;
      workTimer = 180;
      triggerInsideView(target.x, target.z, target.rot);
    }
  }
  
  void draw() {
    pushMatrix();
    translate(x, -45, z);
    
    if (workTimer > 0) {
      rotateY(machines.get(targetIndex).rot + PI);
    } else if (!isComplete) {
      rotateY(atan2(machines.get(targetIndex).opX - x, machines.get(targetIndex).opZ - z));
    }
    
    stroke(20);
    strokeWeight(1);
    
    // Body
    fill(200, 60, 60);
    box(25, 45, 18);
    
    // Hard Hat
    translate(0, -32, 0);
    fill(255, 200, 0);
    sphereDetail(12);
    sphere(14);
    
    // Animated Arms
    fill(200, 60, 60);
    float armAngle = sin(animPhase) * 0.6;
    
    pushMatrix();
    translate(-18, 20, 0);
    rotateX(armAngle);
    box(10, 35, 10);
    popMatrix();
    
    pushMatrix();
    translate(18, 20, 0);
    rotateX(-armAngle);
    box(10, 35, 10);
    popMatrix();
    
    popMatrix();
  }
}
