// =====================================================================
// PERFORMANCE RATING & CORRELATION SIMULATOR (WITH FORMULA REFERENCE SCREEN)
// Features: 4 Interactive Modes (Simulation, Tables, Graph, Formulas),
//           Stickman Animation, Matrix Calculations, Regression Analysis.
// =====================================================================

int currentMode = 1; // 1: Simulation, 2: Tables, 3: Graph, 4: Formulas
boolean useActualData = true;

DataSet actualSet;
DataSet randomSet;
DataSet currentSet;

// Simulation Variables
int activeWalkerIndex = 0;
boolean isWalking = false;
float walkTimer = 0.0;
float walkDuration = 0.0;
float startX, endX;

void setup() {
  size(1280, 720, P3D);
  setWindowFrameTitle("Performance Rating & Correlation Simulator");
  
  // Set physical boundary references for the screen
  startX = width * 0.15;
  endX = width * 0.85;
  
  // Initialize Datasets
  actualSet = new DataSet("ACTUAL NOTEBOOK DATA");
  loadActualData(actualSet);
  actualSet.calculateDerivedValues();
  
  randomSet = new DataSet("RANDOM SIMULATED DATA");
  generateRandomData(randomSet);
  randomSet.calculateDerivedValues();
  
  currentSet = actualSet;
}

void draw() {
  background(240, 245, 250);
  drawTopNav();
  
  if (currentMode == 1) {
    drawSimulationScreen();
  } else if (currentMode == 2) {
    drawTablesScreen();
  } else if (currentMode == 3) {
    drawGraphScreen();
  } else if (currentMode == 4) {
    drawFormulaScreen();
  }
}

// =========================================================
// UI CONTROLS & NAVIGATION
// =========================================================

void mousePressed() {
  // Navigation Tabs (4 buttons)
  if (checkBtn(20, 20, 140, 40)) currentMode = 1;
  if (checkBtn(170, 20, 140, 40)) currentMode = 2;
  if (checkBtn(320, 20, 140, 40)) currentMode = 3;
  if (checkBtn(470, 20, 140, 40)) currentMode = 4;
  
  // Data Toggle Button (Only active in modes 1, 2, 3)
  if (currentMode != 4 && checkBtn(width - 250, 20, 230, 40)) {
    useActualData = !useActualData;
    currentSet = useActualData ? actualSet : randomSet;
    isWalking = false; 
    activeWalkerIndex = 0;
  }
  
  // Simulation Controls
  if (currentMode == 1) {
    if (checkBtn(width/2 - 100, height - 80, 200, 50) && !isWalking) {
      isWalking = true;
      walkTimer = 0;
      walkDuration = currentSet.records[activeWalkerIndex].observedTime;
    }
  }
}

boolean checkBtn(float x, float y, float w, float h) {
  return mouseX >= x && mouseX <= x + w && mouseY >= y && mouseY <= y + h;
}

void drawTopNav() {
  fill(40, 50, 60);
  rect(0, 0, width, 80);
  
  drawButton(20, 20, 140, 40, "1. Sim", currentMode == 1 ? color(0, 150, 200) : color(80));
  drawButton(170, 20, 140, 40, "2. Tables", currentMode == 2 ? color(0, 150, 200) : color(80));
  drawButton(320, 20, 140, 40, "3. Graph", currentMode == 3 ? color(0, 150, 200) : color(80));
  drawButton(470, 20, 140, 40, "4. Formulas", currentMode == 4 ? color(220, 120, 0) : color(80));
  
  if (currentMode != 4) {
    color toggleCol = useActualData ? color(40, 180, 90) : color(200, 100, 40);
    drawButton(width - 250, 20, 230, 40, "DATA: " + (useActualData ? "ACTUAL" : "RANDOM"), toggleCol);
    
    fill(255);
    textAlign(CENTER, CENTER);
    textSize(20);
    text(currentSet.title, width/2 - 70, 40);
  } else {
    fill(255);
    textAlign(CENTER, CENTER);
    textSize(20);
    text("TIME STUDY CALCULATION REFERENCE GUIDE", width/2 + 100, 40);
  }
}

void drawButton(float x, float y, float w, float h, String label, color bg) {
  fill(bg); stroke(255); strokeWeight(2);
  rect(x, y, w, h, 8);
  fill(255); textAlign(CENTER, CENTER); textSize(15);
  text(label, x + w/2, y + h/2 - 2);
}

// =========================================================
// MODE 1: STICKMAN WALKING SIMULATION
// =========================================================

void drawSimulationScreen() {
  fill(30); textAlign(CENTER, TOP); textSize(24);
  text("TIME STUDY EXPERIMENT: 25 FEET WALKING BOUNDARY", width/2, 100);
  
  float groundY = height/2 + 100;
  stroke(100); strokeWeight(4); line(0, groundY, width, groundY);
  
  stroke(200, 50, 50); strokeWeight(4);
  line(startX, groundY - 150, startX, groundY + 20);
  line(endX, groundY - 150, endX, groundY + 20);
  
  fill(200, 50, 50); textAlign(CENTER, BOTTOM); textSize(18);
  text("START", startX, groundY - 160);
  text("END (25 ft)", endX, groundY - 160);
  
  float stickmanX = startX;
  float walkCycle = 0;
  
  Record rec = currentSet.records[activeWalkerIndex];
  
  if (isWalking) {
    float simSpeedScale = 60.0; 
    float timeIncrement = 1.0 / simSpeedScale; 
    walkTimer += timeIncrement;
    
    if (walkTimer >= walkDuration) {
      walkTimer = walkDuration;
      isWalking = false;
      if (activeWalkerIndex < currentSet.records.length - 1) activeWalkerIndex++;
    }
    
    float progress = walkTimer / walkDuration;
    stickmanX = lerp(startX, endX, progress);
    walkCycle = walkTimer * 15.0; 
  } else if (activeWalkerIndex > 0 && walkTimer == 0) {
    stickmanX = startX;
  }
  
  drawStickman(stickmanX, groundY, walkCycle);
  
  fill(240); stroke(0); strokeWeight(2);
  rect(width/2 - 200, 150, 400, 100, 10);
  fill(0); textAlign(CENTER, CENTER); textSize(20);
  text("Current Walker: " + rec.id, width/2, 175);
  
  if (isWalking) {
    fill(200, 50, 50);
    text("Timer: " + nf(walkTimer, 0, 2) + " sec", width/2, 215);
  } else {
    fill(40, 150, 80);
    text("Observed Time: " + nf(rec.observedTime, 0, 2) + " sec", width/2, 215);
    drawButton(width/2 - 100, height - 80, 200, 50, "WALK " + rec.id, color(50, 120, 200));
  }
}

void drawStickman(float x, float y, float cycle) {
  stroke(0); strokeWeight(4); noFill();
  line(x, y - 80, x, y - 40);
  fill(255); ellipse(x, y - 95, 30, 30);
  float swing = sin(cycle) * 20;
  line(x, y - 40, x - swing, y);
  line(x, y - 40, x + swing, y);
  line(x, y - 70, x + swing, y - 40);
  line(x, y - 70, x - swing, y - 40);
}

// =========================================================

void drawGraphScreen() {
  fill(30); textAlign(CENTER, TOP); textSize(26);
  text("SCATTER PLOT & REGRESSION LINE", width/2, 100);
  
  float gX = 200; float gY = height - 150;
  float gW = 700; float gH = 400;
  
  stroke(0); strokeWeight(3);
  line(gX, gY, gX + gW, gY); 
  line(gX, gY, gX, gY - gH); 
  
  fill(0); textAlign(CENTER, TOP); textSize(18);
  text("Average Observed Rating (X) %", gX + gW/2, gY + 40);
  
  pushMatrix();
  translate(gX - 60, gY - gH/2);
  rotate(-PI/2);
  textAlign(CENTER, BOTTOM);
  text("Actual Rating (Y) %", 0, 0);
  popMatrix();
  
  float minVal = 80; float maxVal = 120;
  
  stroke(200); strokeWeight(1); textAlign(RIGHT, CENTER);
  for (int i=80; i<=120; i+=5) {
    float yMap = map(i, minVal, maxVal, gY, gY - gH);
    line(gX, yMap, gX + gW, yMap);
    fill(0); text(i, gX - 10, yMap);
  }
  textAlign(CENTER, TOP);
  for (int i=80; i<=120; i+=5) {
    float xMap = map(i, minVal, maxVal, gX, gX + gW);
    line(xMap, gY, xMap, gY - gH);
    fill(0); text(i, xMap, gY + 10);
  }
  
  fill(200, 50, 50); stroke(150, 0, 0); strokeWeight(2);
  for (int i = 0; i < 5; i++) {
    Record r = currentSet.records[i];
    float px = map(r.X, minVal, maxVal, gX, gX + gW);
    float py = map(r.Y, minVal, maxVal, gY, gY - gH);
    ellipse(px, py, 15, 15);
    
    fill(0); textAlign(LEFT, BOTTOM); textSize(14);
    text(r.id, px + 10, py - 10);
    fill(200, 50, 50);
  }
  
  int n = 5;
  float m = (n * currentSet.sumXY - currentSet.sumX * currentSet.sumY) / 
            (n * currentSet.sumX2 - pow(currentSet.sumX, 2));
  float c = (currentSet.sumY - m * currentSet.sumX) / n;
  
  float startLineX = minVal;
  float startLineY = m * startLineX + c;
  float endLineX = maxVal;
  float endLineY = m * endLineX + c;
  
  float p1x = map(startLineX, minVal, maxVal, gX, gX + gW);
  float p1y = map(startLineY, minVal, maxVal, gY, gY - gH);
  float p2x = map(endLineX, minVal, maxVal, gX, gX + gW);
  float p2y = map(endLineY, minVal, maxVal, gY, gY - gH);
  
  stroke(0, 100, 200); strokeWeight(4);
  line(p1x, p1y, p2x, p2y);
  
  fill(245, 245, 220); stroke(0); strokeWeight(2);
  rect(width - 320, 150, 280, 200, 10);
  fill(0); textAlign(LEFT, TOP); textSize(18);
  text("STATISTICS:", width - 300, 170);
  textSize(16);
  text("n = 5 observations", width - 300, 210);
  text("Slope (m) = " + nf(m, 0, 3), width - 300, 240);
  text("Intercept (c) = " + nf(c, 0, 3), width - 300, 270);
  
  fill(0, 120, 0); textSize(20);
  text("Pearson's r = " + nf(currentSet.r, 0, 3), width - 300, 310);
}

// =========================================================
// DATA STRUCTURES & MATH LOGIC
// =========================================================

class Record {
  String id;
  String keeper;
  float observedTime;
  float[] ratings = new float[5];
  
  float X, Y, basicTime, XY, X2, Y2;
  
  Record(String _id, String _keeper, float _time, float[] _r) {
    id = _id; keeper = _keeper; observedTime = _time; ratings = _r;
  }
}

class DataSet {
  String title;
  Record[] records = new Record[5];
  float avgBasicTimeTotal;
  float sumX, sumY, sumXY, sumX2, sumY2;
  float r; 
  
  DataSet(String _t) { title = _t; }
  
  void calculateDerivedValues() {
    float totalBasicTime = 0;
    
    for (int i=0; i<5; i++) {
      float sumRatings = 0;
      for (int j=0; j<5; j++) sumRatings += records[i].ratings[j];
      
      records[i].X = sumRatings / 5.0; 
      records[i].basicTime = records[i].observedTime * (records[i].X / 100.0);
      totalBasicTime += records[i].basicTime;
    }
    
    avgBasicTimeTotal = totalBasicTime / 5.0;
    
    sumX = 0; sumY = 0; sumXY = 0; sumX2 = 0; sumY2 = 0;
    
    for (int i=0; i<5; i++) {
      records[i].Y = (avgBasicTimeTotal * 100.0) / records[i].observedTime; 
      
      records[i].XY = records[i].X * records[i].Y;
      records[i].X2 = records[i].X * records[i].X;
      records[i].Y2 = records[i].Y * records[i].Y;
      
      sumX += records[i].X;
      sumY += records[i].Y;
      sumXY += records[i].XY;
      sumX2 += records[i].X2;
      sumY2 += records[i].Y2;
    }
    
    float n = 5;
    float numerator = (n * sumXY) - (sumX * sumY);
    float denomPart1 = sqrt((n * sumX2) - pow(sumX, 2));
    float denomPart2 = sqrt((n * sumY2) - pow(sumY, 2));
    r = numerator / (denomPart1 * denomPart2);
  }
}

void loadActualData(DataSet ds) {
  ds.records[0] = new Record("A", "B", 8.0, new float[]{100, 95, 105, 102, 98});
  ds.records[1] = new Record("B", "C", 7.5, new float[]{105, 100, 110, 108, 103});
  ds.records[2] = new Record("C", "D", 9.0, new float[]{90, 88, 92, 95, 93});
  ds.records[3] = new Record("D", "E", 8.5, new float[]{98, 102, 97, 100, 101});
  ds.records[4] = new Record("E", "F", 7.8, new float[]{102, 105, 100, 103, 106});
}

void generateRandomData(DataSet ds) {
  String[] ids = {"A", "B", "C", "D", "E"};
  String[] keepers = {"B", "C", "D", "E", "F"};
  for (int i=0; i<5; i++) {
    float rTime = random(7.0, 9.5); 
    float[] rRatings = new float[5];
    float baseRating = map(rTime, 9.5, 7.0, 85, 110); 
    for (int j=0; j<5; j++) {
      rRatings[j] = round(baseRating + random(-8, 8));
    }
    ds.records[i] = new Record(ids[i], keepers[i], round(rTime*10)/10.0, rRatings);
  }
}

void setWindowFrameTitle(String title) {
  try {
    java.lang.reflect.Method m = getClass().getMethod("windowTitle", String.class);
    m.invoke(this, title);
  } catch (Exception e1) {
    try {
      if (surface != null) {
        java.lang.reflect.Method m = surface.getClass().getMethod("setTitle", String.class);
        m.invoke(surface, title);
      }
    } catch (Exception e2) {}
  }
}
